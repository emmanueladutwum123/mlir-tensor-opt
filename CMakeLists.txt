cmake_minimum_required(VERSION 3.20)
project(mlir-tensor-opt LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()

find_package(MLIR REQUIRED CONFIG)
message(STATUS "Using MLIR ${LLVM_PACKAGE_VERSION} from ${MLIR_DIR}")

list(APPEND CMAKE_MODULE_PATH "${MLIR_CMAKE_DIR}" "${LLVM_CMAKE_DIR}")
include(TableGen)
include(AddLLVM)
include(AddMLIR)

# Match how LLVM itself was built. It has exceptions off; mixing the two
# settings in one binary is undefined.
if(NOT LLVM_ENABLE_EH)
  add_compile_options(-fno-exceptions)
endif()
if(NOT LLVM_ENABLE_RTTI)
  add_compile_options(-fno-rtti)
endif()

# Warnings are errors for code in this repository. LLVM and MLIR headers come
# in as SYSTEM includes, so a warning in upstream code cannot break the build.
# Code generated by mlir-tblgen from *our* .td files is not exempt, with one
# carve-out copied from LLVM's own build: -Wunused-parameter, which the
# generated accessors and property hooks trip by design. Rejected: including
# build/include as SYSTEM instead, which would silence every warning in the
# generated code rather than that one.
add_compile_options(-Wall -Wextra -Wno-unused-parameter -Werror)

include_directories(SYSTEM ${LLVM_INCLUDE_DIRS} ${MLIR_INCLUDE_DIRS})
include_directories(${PROJECT_SOURCE_DIR}/include ${PROJECT_BINARY_DIR}/include)
separate_arguments(LLVM_DEFINITIONS_LIST NATIVE_COMMAND ${LLVM_DEFINITIONS})
add_definitions(${LLVM_DEFINITIONS_LIST})

add_subdirectory(include/TensorOpt)
add_subdirectory(lib/TensorOpt)
add_subdirectory(tools/topt-opt)
add_subdirectory(test)
